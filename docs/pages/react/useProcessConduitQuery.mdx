# useProcessConduitQuery

React mutation hook for processing a query on a conduit.

## Import

```ts
import { useProcessConduitQuery } from 'railnet-sdk/react'
```

## Usage

```tsx
import { useProcessConduitQuery } from 'railnet-sdk/react'
import { ConduitMode } from 'railnet-sdk'

function Process() {
  const { mutate, isPending } = useProcessConduitQuery()

  return (
    <button
      disabled={isPending}
      onClick={() =>
        mutate({
          conduit: '0x...',
          account: '0x...',
          query: {
            owner: '0x...',
            receiver: '0x...',
            input: [{ asset: '0x...', value: 100n }],
            output: [],
            mode: ConduitMode.DEPOSIT,
            salt: '0x...',
            data: '0x',
          },
        })
      }
    >
      Process Query
    </button>
  )
}
```

## Returns

```ts
UseMutationResult<ProcessConduitQueryReturnType, Error, ProcessConduitQueryParameters & { account: Address }>
```

Mutation result. Call `mutate` or `mutateAsync` with [`processConduitQuery`](/actions/processConduitQuery) parameters.
