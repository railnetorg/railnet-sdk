# conduitInfoQueryOptions

Query options for fetching conduit information.

## Import

```ts
import { conduitInfoQueryOptions } from '@railnetorg/railnet-sdk/react'
```

## Usage

```ts
import { useQuery } from '@tanstack/react-query'
import { usePublicClient } from 'wagmi'
import { conduitInfoQueryOptions } from '@railnetorg/railnet-sdk/react'

function ConduitInfo() {
  const client = usePublicClient()

  const { data } = useQuery({
    ...conduitInfoQueryOptions(client, {
      conduit: '0xA0Cf798816D4b9b9866b5330EEa46a18382f251e',
    }),
    staleTime: 30_000, // custom stale time
  })
}
```

### Prefetching

```ts
import { QueryClient } from '@tanstack/react-query'
import { createPublicClient, http } from 'viem'
import { base } from 'viem/chains'
import { conduitInfoQueryOptions } from '@railnetorg/railnet-sdk/react'

const queryClient = new QueryClient()

const client = createPublicClient({
  chain: base,
  transport: http(),
})

await queryClient.prefetchQuery(
  conduitInfoQueryOptions(client, {
    conduit: '0xA0Cf798816D4b9b9866b5330EEa46a18382f251e',
  }),
)
```

## Returns

```ts
QueryOptions<ConduitInfo, ReadContractErrorType, ConduitInfo, ConduitInfoQueryKey>
```

Options object containing `queryFn` and `queryKey`.

## Parameters

### client

- **Type:** `Client<Transport, Chain | undefined> | undefined`

A viem public client.

### parameters

#### conduit

- **Type:** `Address`

The address of the conduit contract.

## Query Key

```ts
['railnet', 'conduitInfo', { chainId, conduit }] as const
```

The hook hands the key back on its result, so the common case needs no builder and no guess about
which chain the client resolved to:

```ts
const { data, queryKey } = useConduitInfo({ conduit: '0x...' })

queryClient.invalidateQueries({ queryKey })
```

Reach for the builder when the hook is not mounted where you invalidate — a mutation's
`onSuccess` in another component, a route loader.

You can use `conduitInfoQueryKey` to construct the query key manually for cache invalidation:

```ts
import { useQueryClient } from '@tanstack/react-query'
import { base } from 'viem/chains'
import { conduitInfoQueryKey } from '@railnetorg/railnet-sdk/react'

const queryClient = useQueryClient()

queryClient.invalidateQueries({
  queryKey: conduitInfoQueryKey(base.id, {
    conduit: '0x...',
  }),
})
```

Every read is scoped to a chain, so the key carries the chain it was filled from. The prefix is
exported too, to invalidate the family across every chain and every parameter set:

```ts
import { conduitInfoQueryPrefix } from '@railnetorg/railnet-sdk/react'

queryClient.invalidateQueries({ queryKey: conduitInfoQueryPrefix })
```
